% The title band ends at 10.65cm and the references band starts at 74.82cm, so
% the two content columns sit between them with a ~0.9cm gap at each end.  The
% last panel of each column is placed with \PosterAutoBlockTo, which stretches
% it down to \PosterColumnBottom; that is what keeps the two columns finishing
% on the same line as their contents change.

\setlength\ColLeft{11.55cm}
\newcommand{\PosterColumnBottom}{73.90cm}

% Right column -- where a Persian reader starts.
